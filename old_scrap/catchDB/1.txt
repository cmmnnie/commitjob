패키지 설치
pip install -r requirements.txt

MySQL에서 실행
CREATE DATABASE catch DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
CREATE USER 'catch_user'@'%' IDENTIFIED BY 'catch_pass';
GRANT ALL PRIVILEGES ON catch.* TO 'catch_user'@'%';
FLUSH PRIVILEGES;

PowerShell
$env:MYSQL_URL = "mysql+pymysql://catch_user:catch_pass@127.0.0.1:3306/catch?charset=utf8mb4"
setx MYSQL_URL "mysql+pymysql://catch_user:catch_pass@127.0.0.1:3306/catch?charset=utf8mb4"
