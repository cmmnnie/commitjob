app.post('/api/scrape-latest-jobs', async (req, res) => {
  console.log('[SCRAPE-JOBS] 최신 공고 스크래핑 요청 받음');

  try {
    // 1. Catch Scraper 서비스 연결 테스트
    console.log('[SCRAPE-JOBS] Catch Scraper 서비스 연결 테스트...');
    console.log('[SCRAPE-JOBS] CATCH_SCRAPER_URL:', CATCH_SCRAPER_URL);

    // 2. Catch Scraper 초기화
    console.log('[SCRAPE-JOBS] Catch Scraper 초기화 중...');
    try {
      await axios.post(`${CATCH_SCRAPER_URL}/api/init`, {}, { timeout: 30000 });
      console.log('[SCRAPE-JOBS] ✅ Catch Scraper 초기화 완료');
    } catch (initError) {
      console.error('[SCRAPE-JOBS] Catch Scraper 초기화 실패:', initError.message);
      if (initError.code === 'ECONNREFUSED') {
        return res.status(503).json({
          success: false,
          message: 'Catch Scraper 서비스에 연결할 수 없습니다. 서비스가 실행 중인지 확인해주세요.'
        });
      }
      return res.status(500).json({
        success: false,
        message: `Catch Scraper 초기화에 실패했습니다: ${initError.message}`
      });
    }

    // 2. Catch.co.kr 로그인
    console.log('[SCRAPE-JOBS] Catch.co.kr 로그인 중...');
    try {
      const loginResponse = await axios.post(`${CATCH_SCRAPER_URL}/api/login`, {
        username: 'test0137',
        password: '#test0808'
      }, { timeout: 30000 });

      if (!loginResponse.data.success) {
        throw new Error('로그인 실패');
      }
      console.log('[SCRAPE-JOBS] ✅ 로그인 완료');
    } catch (loginError) {
      console.error('[SCRAPE-JOBS] 로그인 실패:', loginError.message);
      return res.status(500).json({
        success: false,
        message: 'Catch.co.kr 로그인에 실패했습니다.'
      });
    }

    // 3. 채용공고 페이지로 이동
    console.log('[SCRAPE-JOBS] 채용공고 페이지로 이동 중...');
    try {
      const recruitResponse = await axios.post(`${CATCH_SCRAPER_URL}/api/recruit`, {}, { timeout: 60000 });
      if (!recruitResponse.data.success) {
        console.error('[SCRAPE-JOBS] 채용공고 페이지 이동 실패:', recruitResponse.data);
        throw new Error('채용공고 페이지로 이동할 수 없습니다.');
      }
      console.log('[SCRAPE-JOBS] ✅ 채용공고 페이지 이동 완료');
    } catch (recruitError) {
      console.error('[SCRAPE-JOBS] 채용공고 페이지 이동 오류:', recruitError.message);
      return res.status(500).json({
        success: false,
        message: '채용공고 페이지로 이동할 수 없습니다. 잠시 후 다시 시도해주세요.'
      });
    }

    // 4. IT개발 필터 적용 및 스크래핑 (15건)
    console.log('[SCRAPE-JOBS] IT개발 공고 스크래핑 중...');
    let itJobs = [];

    try {
      // IT개발 필터 적용
      console.log('[SCRAPE-JOBS] IT개발 필터 적용 중...');
      const itFilterResponse = await axios.post(`${CATCH_SCRAPER_URL}/api/filter-it`, {}, { timeout: 30000 });

      if (!itFilterResponse.data.success) {
        console.error('[SCRAPE-JOBS] ❌ IT개발 필터 적용 실패:', itFilterResponse.data);
        throw new Error(`IT개발 필터 적용 실패: ${JSON.stringify(itFilterResponse.data)}`);
      }

      console.log('[SCRAPE-JOBS] ✅ IT개발 필터 적용 완료');

      // IT개발 공고 추출 (15건)
      const itResponse = await axios.get(
        `${CATCH_SCRAPER_URL}/api/extract-first-page-jobs?max_jobs=15`,
        { timeout: 120000 }
      );

      if (!itResponse.data.success) {
        console.error('[SCRAPE-JOBS] ❌ IT개발 공고 추출 실패:', itResponse.data);
        throw new Error(`IT개발 공고 추출 실패: ${JSON.stringify(itResponse.data)}`);
      }

      if (itResponse.data.jobs) {
        itJobs = itResponse.data.jobs.map(job => ({
          ...job,
          category: 'IT'
        }));
        console.log(`[SCRAPE-JOBS] ✅ IT개발 공고 ${itJobs.length}개 추출 완료`);
      } else {
        console.warn('[SCRAPE-JOBS] ⚠️ IT개발 공고 데이터가 비어있음');
      }
    } catch (itError) {
      console.error('[SCRAPE-JOBS] ❌ IT개발 스크래핑 오류:', itError.message);
      console.error('[SCRAPE-JOBS] IT개발 오류 스택:', itError.stack);
    }

    // 5. 빅데이터·AI 필터 적용 및 스크래핑 (15건)
    console.log('[SCRAPE-JOBS] 빅데이터·AI 공고 스크래핑 중...');
    let bigdataJobs = [];

    try {
      // 채용공고 페이지로 다시 이동 (필터 초기화)
      console.log('[SCRAPE-JOBS] 채용공고 페이지로 다시 이동 (필터 초기화)...');
      await axios.post(`${CATCH_SCRAPER_URL}/api/recruit`, {}, { timeout: 60000 });
      console.log('[SCRAPE-JOBS] ✅ 채용공고 페이지 이동 완료');

      // 빅데이터·AI 필터 적용
      console.log('[SCRAPE-JOBS] 빅데이터·AI 필터 적용 중...');
      const bdFilterResponse = await axios.post(`${CATCH_SCRAPER_URL}/api/filter-bigdata-ai`, {}, { timeout: 30000 });

      if (!bdFilterResponse.data.success) {
        console.error('[SCRAPE-JOBS] ❌ 빅데이터·AI 필터 적용 실패:', bdFilterResponse.data);
        throw new Error(`빅데이터·AI 필터 적용 실패: ${JSON.stringify(bdFilterResponse.data)}`);
      }

      console.log('[SCRAPE-JOBS] ✅ 빅데이터·AI 필터 적용 완료');

      // 빅데이터·AI 공고 추출 (15건)
      const bdResponse = await axios.get(
        `${CATCH_SCRAPER_URL}/api/extract-first-page-jobs?max_jobs=15`,
        { timeout: 120000 }
      );

      if (!bdResponse.data.success) {
        console.error('[SCRAPE-JOBS] ❌ 빅데이터·AI 공고 추출 실패:', bdResponse.data);
        throw new Error(`빅데이터·AI 공고 추출 실패: ${JSON.stringify(bdResponse.data)}`);
      }

      if (bdResponse.data.jobs) {
        bigdataJobs = bdResponse.data.jobs.map(job => ({
          ...job,
          category: 'BIGDATA_AI'
        }));
        console.log(`[SCRAPE-JOBS] ✅ 빅데이터·AI 공고 ${bigdataJobs.length}개 추출 완료`);
      } else {
        console.warn('[SCRAPE-JOBS] ⚠️ 빅데이터·AI 공고 데이터가 비어있음');
      }
    } catch (bdError) {
      console.error('[SCRAPE-JOBS] ❌ 빅데이터·AI 스크래핑 오류:', bdError.message);
      console.error('[SCRAPE-JOBS] 빅데이터·AI 오류 스택:', bdError.stack);
    }

    // 6. 스크래핑된 공고 통합
    let scrapedJobs = [...itJobs, ...bigdataJobs];
    console.log(`[SCRAPE-JOBS] 총 ${scrapedJobs.length}개 공고 스크래핑 완료 (IT: ${itJobs.length}개, BIGDATA/AI: ${bigdataJobs.length}개)`);

    // catch.db 폴백 없이 직접 오류 반환
    if (scrapedJobs.length === 0) {
      console.error('[SCRAPE-JOBS] ❌ 스크래핑된 공고가 없습니다.');
      console.error(`[SCRAPE-JOBS] IT 공고: ${itJobs.length}개, 빅데이터·AI 공고: ${bigdataJobs.length}개`);
